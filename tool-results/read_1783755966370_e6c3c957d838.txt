    84→async function searchOpenFDA(query: string): Promise<OpenFDAResult | null> {
    85→  try {
    86→    const searchTerm = encodeURIComponent(query.toLowerCase());
    87→    // Try brand name first, then generic name
    88→    const url = `https://api.fda.gov/drug/label.json?search=openfda.brand_name:"${searchTerm}"+OR+openfda.generic_name:"${searchTerm}"&limit=1`;
    89→
    90→    const res = await fetch(url, {
    91→      headers: { "User-Agent": "MedSnap/1.0" },
    92→      signal: AbortSignal.timeout(10000),
    93→    });
    94→
    95→    if (!res.ok) {
    96→      if (res.status === 404) return null;
    97→      console.error("[openFDA] error:", res.status);
    98→      return null;
    99→    }
   100→
   101→    const json = await res.json();
   102→    const meta = json.meta?.results as { total?: number } | undefined;
   103→    if (!meta?.total || !json.results?.length) return null;
   104→
   105→    const r = json.results[0];
   106→    const openfda = r.openfda || {};
   107→
   108→    return {
   109→      id: `openfda-${Date.now()}`,
   110→      brandName: (openfda.brand_name?.[0] as string) || query,
   111→      genericName: (openfda.generic_name?.[0] as string) || "",
   112→      manufacturer: openfda.manufacturer_name?.[0] as string | undefined,
   113→      purpose: r.purpose || [],
   114→      indications: r.indications_and_usage || [],
   115→      warnings: r.warnings || [],
   116→      adverseReactions: r.adverse_reactions || [],
   117→      dosageAndAdministration: r.dosage_and_administration || [],
   118→      activeIngredients: r.active_ingredient || openfda.generic_name || [],
   119→      inactiveIngredients: r.inactive_ingredient || [],
   120→      storage: r.storage_and_handling || [],
   121→      pregnancy: r.pregnancy_or_breast_feeding?.[0] as string | undefined,
   122→      pregnancyCategory: (openfda.pregnancy_category?.[0] as string) || undefined,
   123→      doNotUse: r.do_not_use || [],
   124→      askDoctor: r.ask_doctor || [],
   125→      askDoctorOrPharmacist: r.ask_doctor_or_pharmacist || [],
   126→      stopUse: r.stop_use || [],
   127→      drugInteractions: r.drug_interactions || [],
   128→      substanceName: openfda.substance_name?.[0] as string | undefined,
   129→      substanceCode: openfda.substance_code?.[0] as string | undefined,
   130→      pharmacology: r.clinical_pharmacology?.[0] as string | undefined,
   131→      clinicalPharmacology: r.clinical_pharmacology?.[0] as string | undefined,
   132→      mechanismOfAction: r.mechanism_of_action?.[0] as string | undefined,
   133→      contraindications: r.contraindications || [],
   134→    };
   135→  } catch (err) {
   136→    console.error("[openFDA] fetch failed:", err);
   137→    return null;
   138→  }
   139→}

   141→// ---------- 2. RxNorm API ----------
   142→
   143→async function searchRxNorm(query: string): Promise<RxNormResult | null> {
   144→  try {
   145→    const url = `https://rxnav.nlm.nih.gov/REST/drugs.json?name=${encodeURIComponent(query)}`;
   146→    const res = await fetch(url, {
   147→      headers: { "User-Agent": "MedSnap/1.0" },
   148→      signal: AbortSignal.timeout(10000),
   149→    });
   150→    if (!res.ok) return null;
   151→    const json = await res.json();
   152→    const group = json.drugGroup;
   153→    if (!group?.conceptGroup) return null;
   154→
   155→    for (const cg of group.conceptGroup) {
   156→      if (cg.conceptProperties) {
   157→        for (const cp of cg.conceptProperties) {
   158→          if (cp.name) {
   159→            return {
   160→              name: cp.name,
   161→              rxcui: cp.rxcui,
   162→              synonym: cp.synonym,
   163→              tty: cp.tty,
   164→            };
   165→          }
   166→        }
   167→      }
   168→    }
   169→    return null;
   170→  } catch (err) {
   171→    console.error("[RxNorm] fetch failed:", err);
   172→    return null;
   173→  }
   174→}

   176→/**
   177→ * Fuzzy search RxNorm — uses approximate matching for misspellings.
   178→ * This catches typos like "hydrtocortisone" → "hydrocortisone".
   179→ */
   180→async function searchRxNormApproximate(query: string): Promise<RxNormResult | null> {
   181→  try {
   182→    // RxNorm approximate match API — returns similar-sounding drug names
   183→    const url = `https://rxnav.nlm.nih.gov/REST/spellingsuggestions.json?name=${encodeURIComponent(query)}`;
   184→    const res = await fetch(url, {
   185→      headers: { "User-Agent": "MedSnap/1.0" },
   186→      signal: AbortSignal.timeout(10000),
   187→    });
   188→    if (!res.ok) return null;
   189→    const json = await res.json();
   190→    const suggestions = json.suggestionGroup?.suggestionList?.suggestion;
   191→    if (!suggestions?.length) return null;
   192→
   193→    // Use the first suggestion to do a proper drug lookup
   194→    const bestSuggestion = suggestions[0];
   195→    return searchRxNorm(bestSuggestion);
   196→  } catch (err) {
   197→    console.error("[RxNorm approximate] fetch failed:", err);
   198→    return null;
   199→  }
   200→}

   202→/**
   203→ * Search RxNorm by active ingredient — finds all drugs with that ingredient.
   204→ */
   205→async function searchRxNormByIngredient(ingredient: string): Promise<RxNormResult | null> {
   206→  try {
   207→    // Use the approximate match endpoint with ingredient search
   208→    const url = `https://rxnav.nlm.nih.gov/REST/ApproximateMatch?term=${encodeURIComponent(ingredient)}&maxEntries=1`;
   209→    const res = await fetch(url, {
   210→      headers: { "User-Agent": "MedSnap/1.0" },
   211→      signal: AbortSignal.timeout(10000),
   212→    });
   213→    if (!res.ok) return null;
   214→    const json = await res.json();
   215→    const candidates = json.approximateGroup?.candidate;
   216→    if (!candidates?.length) return null;
   217→
   218→    const c = candidates[0];
   219→    if (c.rxcui) {
   220→      return {
   221→        name: c.name || ingredient,
   222→        rxcui: c.rxcui,
   223→        tty: c.tty,
   224→      };
   225→    }
   226→    return null;
   227→  } catch (err) {
   228→    console.error("[RxNorm ingredient] fetch failed:", err);
   229→    return null;
   230→  }
   231→}

   233→// ---------- 3. RxNorm Drug Interactions ----------
   234→
   235→async function getRxNormInteractions(rxcui: string): Promise<Interaction[]> {
   236→  try {
   237→    const url = `https://rxnav.nlm.nih.gov/REST/interaction/interaction.json?rxcui=${rxcui}`;
   238→    const res = await fetch(url, {
   239→      headers: { "User-Agent": "MedSnap/1.0" },
   240→      signal: AbortSignal.timeout(10000),
   241→    });
   242→    if (!res.ok) return [];
   243→    const json = await res.json();
   244→    const interactionTypeGroup = json.interactionTypeGroup;
   245→    if (!interactionTypeGroup) return [];
   246→
   247→    const interactions: Interaction[] = [];
   248→    for (const group of interactionTypeGroup) {
   249→      if (!group.interactionType) continue;
   250→      for (const type of group.interactionType) {
   251→        if (!type.interactionPair) continue;
   252→        for (const pair of type.interactionPair) {
   253→          const drug = pair.interactionConcept?.[1]?.minConceptItem?.name || "";
   254→          const desc = pair.description || "";
   255→          const severity = pair.severity || "";
   256→          if (drug && desc) {
   257→            interactions.push({
   258→              with: drug,
   259→              severity: severity.toLowerCase().includes("high") || severity.toLowerCase().includes("contraindicated") ? "avoid" : "caution",
   260→              note: desc.slice(0, 300),
   261→            });
   262→          }
   263→        }
   264→      }
   265→    }
   266→    return interactions.slice(0, 8);
   267→  } catch (err) {
   268→    console.error("[RxNorm interactions] fetch failed:", err);
   269→    return [];
   270→  }
   271→}

   289→
   290→    const img = imgs[0];
   291→    return {
   292→      name: img.name || "",
   293→      rxcui: img.rxcui || "",
   294→      imprint: img.imprint?.[0],
   295→      color: img.color,
   296→      shape: img.shape,
   297→      imageUrl: img.imageUrl,
   298→      size: img.size,

   309→  try {
   310→    const url = `https://dailymed.nlm.nih.gov/dailymed/services/v2/drugnames.json?drug_name=${encodeURIComponent(query)}&pagesize=1`;
   311→    const res = await fetch(url, {
   312→      headers: { "User-Agent": "MedSnap/1.0" },
   313→      signal: AbortSignal.timeout(10000),
   314→    });
   315→    if (!res.ok) return null;
   316→    const json = await res.json();
   317→    const items = json.data;
   318→    if (!items?.length) return null;
   319→
   320→    const item = items[0];
   321→    return {
   322→      title: item.drug_name || item.title || query,
   323→      url: `https://dailymed.nlm.nih.gov/dailymed/search.cfm?labeltype=all&query=${encodeURIComponent(query)}`,
   324→      setid: item.setid,
   325→    };
   326→  } catch (err) {
   327→    console.error("[DailyMed] fetch failed:", err);
   328→    return null;
   329→  }

   332→// ---------- Main identification function ----------
   333→
   334→export async function identifyFromVerifiedSources(
   335→  params: IdentifyParams
   336→): Promise<MedicineResult> {
   337→  const { query, shape, color } = params;
   338→  const cleanQuery = query.trim();
   339→
   340→  if (!cleanQuery) {
   341→    return buildNotFoundResult("Please enter a medicine name or imprint code.");
   342→  }
   343→
   344→  const sources: { label: string; url?: string }[] = [];
   345→  const isImprint = /^\d+$/.test(cleanQuery) || cleanQuery.length <= 6;
   346→
   347→  // Step 1: If it looks like an imprint code, try RxImage first
   348→  let rxImage: RxImageResult | null = null;
   349→  if (isImprint) {
   350→    rxImage = await searchRxImage(cleanQuery, color, shape);
   351→    if (rxImage?.name) {
   352→      sources.push({
   353→        label: `RxImage (NIH) — Pill match`,
   354→        url: `https://rximage.nlm.nih.gov/docs/RxImageSearch.html`,
   355→      });
   356→    }
   357→  }
   358→
   359→  // Step 2: MULTI-STRATEGY SEARCH IN PARALLEL
   360→  const searchTerm = rxImage?.name || cleanQuery;
   361→
   362→  const [openfda, rxnormExact, rxnormFuzzy, dailymed] = await Promise.all([
   363→    searchOpenFDA(searchTerm),
   364→    searchRxNorm(searchTerm),
   365→    searchRxNormApproximate(searchTerm),
   366→    searchDailyMed(searchTerm),
   367→  ]);
   368→
   369→  const rxnorm = rxnormExact || rxnormFuzzy;
   370→
   371→  if (openfda) {
   372→    sources.push({
   373→      label: "openFDA Drug Label (FDA)",
   374→      url: `https://open.fda.gov/data/downloads/`,
   375→    });
   376→  }
   377→
   378→  if (rxnorm) {
   379→    sources.push({
   380→      label: rxnormFuzzy && !rxnormExact ? "RxNorm (NIH) — fuzzy match" : "RxNorm (NIH)",
   381→      url: `https://mor.nlm.nih.gov/RxClass/search?query=${encodeURIComponent(searchTerm)}&searchBy=drug`,
   382→    });
   383→  }
   384→
   385→  // Step 3: Get drug interactions from RxNorm (if we have an rxcui)
   386→  let interactions: Interaction[] = [];
   387→  if (rxnorm?.rxcui) {
   388→    interactions = await getRxNormInteractions(rxnorm.rxcui);
   389→    if (interactions.length > 0) {
   390→      sources.push({
   391→        label: "RxNorm Interactions (NIH)",
   392→        url: `https://rxnav.nlm.nih.gov/REST/interaction/interaction.json?rxcui=${rxnorm.rxcui}`,
   393→      });
   394→    }
   395→  }
   396→
   397→  if (dailymed) {
   398→    sources.push({ label: "DailyMed (NIH)", url: dailymed.url });
   399→  }
   400→
   401→  // If we found nothing at all, try ingredient-based search as last resort
   402→  if (!openfda && !rxnorm && !dailymed && !rxImage) {
   403→    // Try searching by ingredient (extract from query — might be a partial name)
   404→    const ingredientResult = await searchRxNormByIngredient(cleanQuery);
   405→    if (ingredientResult) {
   406→      const [openfda2, dailymed2] = await Promise.all([
   407→        searchOpenFDA(ingredientResult.name),
   408→        searchDailyMed(ingredientResult.name),
   409→      ]);
   410→      if (openfda2 || dailymed2) {
   411→        sources.push({
   412→          label: "RxNorm (NIH) — ingredient match",
   413→          url: `https://rxnav.nlm.nih.gov/REST/ApproximateMatch?term=${encodeURIComponent(cleanQuery)}`,
   414→        });
   415→        if (dailymed2) sources.push({ label: "DailyMed (NIH)", url: dailymed2.url });
   416→        // Reassemble with these results
   417→        const result = assembleResult(
   418→          openfda2,
   419→          ingredientResult,
   420→          rxImage,
   421→          dailymed2,
   422→          interactions,
   423→          sources,
   424→          cleanQuery,
   425→          shape,
   426→          color
   427→        );
   428→        return result;
   429→      }
   430→    }
   431→    return buildNotFoundResult(
   432→      `No verified information found for "${cleanQuery}". Try the brand name (e.g. "Tylenol") or generic name (e.g. "acetaminophen").`
   433→    );
   434→  }
   435→
   436→  // Assemble the MedicineResult
   437→  return assembleResult(openfda, rxnorm, rxImage, dailymed, interactions, sources, cleanQuery, shape, color);
   438→}

   440→/**
   441→ * Assemble the final MedicineResult from all the gathered data.
   442→ */
   443→function assembleResult(
   444→  openfda: OpenFDAResult | null,
   445→  rxnorm: RxNormResult | null,
   446→  rxImage: RxImageResult | null,
   447→  dailymed: DailyMedResult | null,
   448→  interactions: Interaction[],
   449→  sources: { label: string; url?: string }[],
   450→  cleanQuery: string,
   451→  shape?: string,
   452→  color?: string
   453→): MedicineResult {
   454→  const brandName = openfda?.brandName || rxnorm?.name || dailymed?.title || cleanQuery;
   455→  const genericName = openfda?.genericName || rxnorm?.name || "";
   456→
   457→  const strengthMatch = cleanQuery.match(/(\d+(?:\.\d+)?)\s*(mg|mcg|ml|g|%)/i);
   458→  const strengthValue = strengthMatch?.[1] || extractStrengthFromLabel(openfda) || "";
   459→  const strengthUnit = strengthMatch?.[2]?.toLowerCase() || "";
   460→
   461→  return {
   462→    id: `verified-${Date.now()}`,
   463→    brandName,
   464→    genericName: genericName || NOT_FOUND,
   465→    manufacturer: openfda?.manufacturer,
   466→    strengthValue: strengthValue || "?",
   467→    strengthUnit,
   468→    strengthDisplay: strengthValue ? `${strengthValue} ${strengthUnit}`.trim() : "See label",
   469→    form: detectForm(openfda, cleanQuery, rxImage),
   470→    packageSize: undefined,
   471→    usedFor: extractUsedFor(openfda),
   472→    activeIngredients: openfda?.activeIngredients?.length
   473→      ? openfda.activeIngredients.slice(0, 6)
   474→      : genericName
   475→        ? [genericName]
   476→        : [NOT_FOUND],
   477→    commonSideEffects: extractCommonSideEffects(openfda),
   478→    seriousSideEffects: extractSeriousSideEffects(openfda),
   479→    interactions: interactions.length > 0
   480→      ? interactions
   481→      : extractInteractionsFromLabel(openfda),
   482→    whoShouldAvoid: extractWhoShouldAvoid(openfda),
   483→    storageInstructions: openfda?.storage?.[0] || NOT_FOUND,
   484→    confidence: openfda ? "high" : rxnorm ? "medium" : "low",
   485→    matchNote: buildMatchNote(rxImage, openfda, rxnorm, cleanQuery),
   486→    sources: sources.length > 0 ? sources : [{ label: "Verified government databases" }],
   487→    imprint: rxImage?.imprint || (cleanQuery.length <= 6 ? cleanQuery : undefined),
   488→    composition: buildComposition(openfda),
   489→    dietaryAdvice: extractDosageAdvice(openfda),
   490→    howItWorks: openfda?.mechanismOfAction || openfda?.pharmacology || undefined,
   491→    drugClass: extractDrugClass(openfda, rxnorm),
   492→    mechanismOfAction: openfda?.mechanismOfAction || openfda?.pharmacology || undefined,
   493→    metabolism: extractFromClinical(openfda, /metabol/i),
   494→    excretion: extractFromClinical(openfda, /excret/i),
   495→    halfLife: extractFromClinical(openfda, /half.?life/i),
   496→    onsetOfAction: extractFromClinical(openfda, /onset/i),
   497→    durationOfAction: extractFromClinical(openfda, /duration/i),
   498→    pregnancyCategory: openfda?.pregnancyCategory || extractPregnancyCategory(openfda),
   499→    whatToDoIfMissed: extractMissedDose(openfda),
   500→    overdoseSymptoms: extractOverdoseSymptoms(openfda),
   501→    relatedMedicines: extractRelatedMedicines(openfda, rxnorm),
   502→  };
   503→}

   505→// ---------- Search function (for manual search screen) ----------
   506→
   507→export async function searchVerifiedSources(query: string): Promise<MedicineResult[]> {
   508→  const cleanQuery = query.trim();
   509→  if (!cleanQuery) return [];
   510→
   511→  try {
   512→    const searchTerm = encodeURIComponent(cleanQuery.toLowerCase());
   513→    const url = `https://api.fda.gov/drug/label.json?search=openfda.brand_name:"${searchTerm}"+OR+openfda.generic_name:"${searchTerm}"&limit=5`;
   514→    const res = await fetch(url, {
   515→      headers: { "User-Agent": "MedSnap/1.0" },
   516→      signal: AbortSignal.timeout(10000),
   517→    });
   518→
   519→    if (res.ok) {
   520→      const json = await res.json();
   521→      if (json.results?.length) {
   522→        return json.results.slice(0, 5).map((r: any, i: number) => {
   523→          const openfda = r.openfda || {};
   524→          return {
   525→            id: `openfda-search-${i}-${Date.now()}`,
   526→            brandName: openfda.brand_name?.[0] || cleanQuery,
   527→            genericName: openfda.generic_name?.[0] || "",
   528→            manufacturer: openfda.manufacturer_name?.[0],
   529→            strengthValue: "?",
   530→            strengthUnit: "",
   531→            strengthDisplay: "See label",
   532→            form: "unknown" as const,
   533→            usedFor: (r.purpose?.slice(0, 3) || ["See full label"]).map((s: string) => cleanLabelText(s)),
   534→            activeIngredients: (r.active_ingredient?.slice(0, 3) || openfda.generic_name || [cleanQuery]).map((s: string) => cleanLabelText(s)),
   535→            commonSideEffects: (r.adverse_reactions?.slice(0, 3) || ["See full label"]).map((s: string) => cleanLabelText(s)),
   536→            seriousSideEffects: (r.warnings?.slice(0, 3) || []).map((s: string) => cleanLabelText(s)),
   537→            interactions: [],
   538→            whoShouldAvoid: [],
   539→            storageInstructions: r.storage_and_handling?.[0] || NOT_FOUND,
   540→            confidence: "high" as const,
   541→            matchNote: `Matched from openFDA drug label database`,
   542→            sources: [
   543→              { label: "openFDA Drug Label (FDA)", url: `https://open.fda.gov/data/downloads/` },
   544→              { label: "DailyMed (NIH)", url: `https://dailymed.nlm.nih.gov/dailymed/search.cfm?query=${searchTerm}` },
   545→            ],
   546→          } as MedicineResult;
   547→        });
   548→      }
   549→    }
   550→  } catch (err) {
   551→    console.error("[searchVerifiedSources] openFDA failed:", err);
   552→  }
   553→
   554→  try {
   555→    const rxnorm = await searchRxNorm(cleanQuery);
   556→    // If exact didn't work, try fuzzy
   557→    const rxnormResult = rxnorm || (await searchRxNormApproximate(cleanQuery));
   558→    if (rxnormResult) {
   559→      return [{
   560→        id: `rxnorm-${Date.now()}`,
   561→        brandName: rxnormResult.name,
   562→        genericName: rxnormResult.name,
   563→        strengthValue: "?",
   564→        strengthUnit: "",
   565→        strengthDisplay: "See label",
   566→        form: "unknown",
   567→        usedFor: ["See full drug information"],
   568→        activeIngredients: [rxnormResult.name],
   569→        commonSideEffects: [NOT_FOUND],
   570→        seriousSideEffects: [NOT_FOUND],
   571→        interactions: [],
   572→        whoShouldAvoid: [],
   573→        storageInstructions: NOT_FOUND,
   574→        confidence: "medium",
   575→        matchNote: rxnorm ? "Matched from RxNorm (NIH) drug database" : "Fuzzy matched from RxNorm (NIH)",
   576→        sources: [
   577→          { label: rxnorm ? "RxNorm (NIH)" : "RxNorm (NIH) — fuzzy match", url: `https://mor.nlm.nih.gov/RxClass/search?query=${encodeURIComponent(cleanQuery)}&searchBy=drug` },
   578→          { label: "DailyMed (NIH)", url: `https://dailymed.nlm.nih.gov/dailymed/search.cfm?query=${encodeURIComponent(cleanQuery)}` },
   579→        ],
   580→      }];
   581→    }
   582→  } catch (err) {
   583→    console.error("[searchVerifiedSources] RxNorm failed:", err);
   584→  }
   585→
   586→  return [];
   587→}

   591→/**
   592→ * Clean raw openFDA label text into readable sentences.
   593→ * Fixes: bullet chars, broken words, weird spacing, OCR-like artifacts.
   594→ */
   595→function cleanLabelText(text: string): string {
   596→  if (!text) return "";
   597→  return text
   598→    // Remove bullet/box-drawing characters
   599→    .replace(/[\u25A0-\u25FF\u2B00-\u2BFF\u2500-\u257F■▶►•‣◦●○]/g, " ")
   600→    // Remove non-printable chars
   601→    .replace(/[^\x20-\x7E\n]/g, " ")
   602→    // Fix common broken-word patterns from FDA labels
   603→    .replace(/\bD\s+irections\b/gi, "Directions")
   604→    .replace(/\bW\s+arnings\b/gi, "Warnings")
   605→    .replace(/\bI\s+ngredients\b/gi, "Ingredients")
   606→    .replace(/\bP\s+urpose\b/gi, "Purpose")
   607→    .replace(/\bU\s+ses\b/gi, "Uses")
   608→    .replace(/\bD\s+osage\b/gi, "Dosage")
   609→    .replace(/\bA\s+ctive\b/gi, "Active")
   610→    .replace(/\bI\s+nactive\b/gi, "Inactive")
   611→    .replace(/\bS\s+torage\b/gi, "Storage")
   612→    .replace(/\bP\s+harmacist\b/gi, "Pharmacist")
   613→    .replace(/\bR\s+elief\b/gi, "Relief")
   614→    .replace(/\bA\s+dverse\b/gi, "Adverse")
   615→    .replace(/\bC\s+ontraindications\b/gi, "Contraindications")
   616→    .replace(/\bP\s+regnancy\b/gi, "Pregnancy")
   617→    .replace(/\bI\s+nteractions\b/gi, "Interactions")
   618→    .replace(/\bO\s+verdose\b/gi, "Overdose")
   619→    .replace(/\bP\s+recautions\b/gi, "Precautions")
   620→    // Fix "500mg" → "500 mg"
   621→    .replace(/(\d)(mg|mcg|ml|g)\b/gi, "$1 $2")
   622→    // Normalize whitespace
   623→    .replace(/\s+/g, " ")
   624→    .trim();
   625→}

   627→/**
   628→ * Split text into clean, readable sentences.
   629→ * Filters out empty/too-short/too-long chunks.
   630→ */
   631→function splitIntoSentences(text: string, minLen = 5, maxLen = 200): string[] {
   632→  if (!text) return [];
   633→  return text
   634→    .split(/[.;\n]/)
   635→    .map((s) => s.trim())
   636→    .filter((s) => s.length >= minLen && s.length <= maxLen)
   637→    .map((s) => {
   638→      // Capitalize first letter
   639→      return s.charAt(0).toUpperCase() + s.slice(1);
   640→    });
   641→}
   642→
   643→/**
   644→ * Capitalize the first letter of a string.
   645→ */
   646→function capitalize(s: string): string {
   647→  if (!s) return s;
   648→  return s.charAt(0).toUpperCase() + s.slice(1);
   649→}

   675→function detectForm(openfda: OpenFDAResult | null, query: string, rxImage?: RxImageResult | null): MedicineResult["form"] {
   676→  // Check RxImage shape first
   677→  if (rxImage?.shape?.[0]) {
   678→    const shape = rxImage.shape[0].toLowerCase();
   679→    if (shape.includes("capsule")) return "capsule";
   680→    if (shape.includes("round") || shape.includes("oval") || shape.includes("oblong")) return "tablet";
   681→  }
   682→  if (!openfda) {
   683→    const q = query.toLowerCase();
   684→    if (q.includes("capsule")) return "capsule";
   685→    if (q.includes("syrup") || q.includes("liquid") || q.includes("solution")) return "syrup";
   686→    if (q.includes("cream") || q.includes("ointment")) return "cream";
   687→    if (q.includes("drop")) return "drops";
   688→    if (q.includes("inhal")) return "inhaler";
   689→    if (q.includes("inject")) return "injection";
   690→    return "tablet";
   691→  }
   692→  const text = [...openfda.activeIngredients, ...openfda.dosageAndAdministration].join(" ").toLowerCase();
   693→  if (text.includes("capsule")) return "capsule";
   694→  if (text.includes("syrup") || text.includes("oral solution") || text.includes("suspension")) return "syrup";
   695→  if (text.includes("cream") || text.includes("ointment")) return "cream";
   696→  if (text.includes("drop")) return "drops";
   697→  if (text.includes("inhal")) return "inhaler";
   698→  if (text.includes("inject")) return "injection";
   699→  return "tablet";
   700→}

   702→function extractStrengthFromLabel(openfda: OpenFDAResult | null): string | undefined {
   703→  if (!openfda?.activeIngredients?.length) return undefined;
   704→  for (const ing of openfda.activeIngredients) {
   705→    const match = ing.match(/(\d+(?:\.\d+)?)\s*(mg|mcg|ml|g|%)/i);
   706→    if (match) return match[1];
   707→  }
   708→  return undefined;
   709→}

   711→function extractUsedFor(openfda: OpenFDAResult | null): string[] {
   712→  if (!openfda) return [NOT_FOUND];
   713→  const out: string[] = [];
   714→  if (openfda.purpose?.length) {
   715→    for (const p of openfda.purpose) {
   716→      const clean = cleanLabelText(p);
   717→      out.push(...splitIntoSentences(clean));
   718→    }
   719→  }
   720→  if (openfda.indications?.length && out.length < 4) {
   721→    const clean = cleanLabelText(openfda.indications[0]);
   722→    out.push(...splitIntoSentences(clean).slice(0, 4 - out.length));
   723→  }
   724→  return out.length > 0 ? out.slice(0, 6) : [NOT_FOUND];
   725→}

   733→function extractSeriousSideEffects(openfda: OpenFDAResult | null): string[] {

   739→  }

   741→    for (const s of openfda.stopUse) {
   742→      const clean = cleanLabelText(s);
   743→      out.push(...splitIntoSentences(clean, 10, 200));
   744→    }
   745→  }
   746→  return out.slice(0, 6);
   747→}

   749→function extractInteractionsFromLabel(openfda: OpenFDAResult | null): Interaction[] {
   750→  if (!openfda?.drugInteractions?.length && !openfda?.askDoctorOrPharmacist?.length) return [];
   751→  const out: Interaction[] = [];
   752→  if (openfda?.drugInteractions?.length) {
   753→    const clean = cleanLabelText(openfda.drugInteractions[0]);
   754→    const lines = splitIntoSentences(clean, 10, 250);
   755→    for (const line of lines.slice(0, 5)) {
   756→      out.push({
   757→        with: "See note",
   758→        severity: "caution",
   759→        note: line,
   760→      });
   761→    }
   762→  }
   763→  return out;
   764→}

   766→function extractWhoShouldAvoid(openfda: OpenFDAResult | null): AvoidFor[] {
   767→  if (!openfda) return [];
   768→  const out: AvoidFor[] = [];
   769→  if (openfda.pregnancy) {
   770→    out.push({ group: "Pregnancy and breastfeeding", reason: cleanLabelText(openfda.pregnancy).slice(0, 250) });
   771→  }
   772→  if (openfda.pregnancyCategory && openfda.pregnancyCategory !== "U") {
   773→    out.push({
   774→      group: "Pregnancy category",
   775→      reason: `FDA Pregnancy Category ${openfda.pregnancyCategory}. Consult your doctor before use during pregnancy.`,
   776→    });
   777→  }
   778→  if (openfda.doNotUse?.length) {
   779→    for (const d of openfda.doNotUse) {
   780→      const clean = cleanLabelText(d);
   781→      const parts = splitIntoSentences(clean, 5, 200);
   782→      for (const p of parts.slice(0, 2)) {
   783→        out.push({ group: "Do not use if", reason: p });
   784→      }
   785→    }
   786→  }
   787→  if (openfda.askDoctor?.length) {
   788→    for (const a of openfda.askDoctor) {
   789→      const clean = cleanLabelText(a);
   790→      const parts = splitIntoSentences(clean, 5, 200);
   791→      for (const p of parts.slice(0, 2)) {
   792→        out.push({ group: "Ask a doctor before use", reason: p });
   793→      }
   794→    }
   795→  }
   796→  if (openfda.contraindications?.length) {
   797→    for (const c of openfda.contraindications) {
   798→      const clean = cleanLabelText(c);
   799→      const parts = splitIntoSentences(clean, 5, 200);
   800→      for (const p of parts.slice(0, 2)) {
   801→        out.push({ group: "Contraindicated for", reason: p });
   802→      }
   803→    }
   804→  }
   805→  return out.slice(0, 6);
   806→}

   814→function buildComposition(openfda: OpenFDAResult | null): string | undefined {
   815→  if (!openfda) return undefined;

   817→  if (openfda.activeIngredients?.length) {

   819→    parts.push(`Active ingredients: ${clean}`);

   821→  if (openfda.inactiveIngredients?.length) {

   823→    parts.push(`Inactive ingredients: ${clean}`);
   824→  }
   825→  return parts.length > 0 ? parts.join(". ") : undefined;
   826→}

   857→function extractPregnancyCategory(openfda: OpenFDAResult | null): string | undefined {
