   686→function ResultCard({
   687→  title,
   688→  icon,
   689→  accent = "default",
   690→  subtitle,
   691→  items,
   692→  onExpand,
   693→}: {
   694→  title: string;
   695→  icon: React.ComponentType<{ className?: string }>;
   696→  accent?: "default" | "safe" | "warn" | "danger";
   697→  subtitle?: string;
   698→  items: string[];
   699→  onExpand?: () => void;
   700→}) {
   701→  if (items.length === 0) return null;
   702→  const preview = items.slice(0, 4);
   703→  const hasMore = items.length > preview.length || Boolean(onExpand);
   704→
   705→  return (
   706→    <Card className="border-border/60 p-5 shadow-soft">
   707→      <SectionHeader title={title} icon={icon} accent={accent} onExpand={hasMore ? onExpand : undefined} />
   708→      {subtitle && <p className="mt-1.5 text-xs text-muted-foreground">{subtitle}</p>}
   709→      <ul className="mt-3 space-y-2">
   710→        {preview.map((item, i) => (
   711→          <li key={i} className="flex items-start gap-2 text-sm">
   712→            <span
   713→              className={cn(
   714→                "mt-1.5 h-1.5 w-1.5 shrink-0 rounded-full",
   715→                accent === "safe" && "bg-safe",
   716→                accent === "warn" && "bg-warn",
   717→                accent === "danger" && "bg-danger",
   718→                accent === "default" && "bg-foreground"
   719→              )}
   720→            />
   721→            <span className="text-foreground/90">{item}</span>
   722→          </li>
   723→        ))}
   724→      </ul>
   725→      {hasMore && items.length > preview.length && (
   726→        <p className="mt-2 text-xs text-muted-foreground">
   727→          + {items.length - preview.length} more
   728→        </p>
   729→      )}
   730→    </Card>
   731→  );
   732→}
